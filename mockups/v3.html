func ProcessTransaction(ctx context.Context, req *TxRequest) (*TxResponse, error) {
  // 1. Verificación de firma criptográfica y token de idempotencia
  if err := crypto.VerifySignature(req.Payload, req.Signature); err != nil {
      return nil, status.Errorf(codes.Unauthenticated, "invalid signature: %v", err)
  }

  // 2. Bloqueo distribuido atómico mediante Redis Cluster
  lock, err := redisClient.AcquireLock(ctx, req.IdempotencyKey, 5*time.Second)
  if err != nil {
      return nil, status.Errorf(codes.ResourceExhausted, "transaction in progress")
  }
  defer lock.Release()

  // 3. Ejecución atómica en PostgreSQL con réplicas de lectura
  txResponse, err := postgresPool.ExecTx(ctx, func(tx *sql.Tx) error {
      return applyDoubleEntryLedger(tx, req)
  })
  
  return txResponse, err
}
